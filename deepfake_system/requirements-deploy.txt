# What Railway installs. Serving only — no training, no evaluation.
#
# Differs from requirements.txt in three ways that matter for the image:
#
#   1. CPU-only torch. The default PyPI wheel on Linux pulls the CUDA
#      build and its bundled nvidia libraries — roughly 2.5 GB for a
#      container that has no GPU. The extra index below is PyTorch's
#      official CPU channel and cuts that to about 200 MB.
#   2. No pandas / scikit-learn / tqdm. Those are training-time only.
#   3. No mediapipe. It is the *second* face-detector fallback; MTCNN
#      via facenet-pytorch is the one the IR specifies and it is here.
#
# ffmpeg and ffprobe are system binaries, not pip packages — they come
# from nixpacks.toml.

--extra-index-url https://download.pytorch.org/whl/cpu

-r requirements-web.txt

torch>=2.2
torchvision>=0.17
timm>=1.0.7
facenet-pytorch>=2.6      # MTCNN, as specified in the IR

yt-dlp>=2024.4            # /api/analyse-url; was imported but never declared
